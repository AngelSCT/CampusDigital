     // CREA UN USUARIO NUEVO EN EL SISTEMA
       ESTA SE DEBE USAR CUANDO UN NUEVO USUARIO SE REGISTRE DESDE OTRO MODULO
       REQUIERE BODY JSON: { nombre, apellido, email, password, telefono }

     // ACTUALIZA LOS DATOS DEL USUARIO CON ID 1
       ESTA SE DEBE USAR CUANDO EL USUARIO EDITE SU PERFIL O UN ADMIN MODIFIQUE SUS DATOS
       SOLO MANDA EN EL BODY LOS CAMPOS QUE QUIERES CAMBIAR

DELETE http://localhost:8000/api/usuarios/1
     // ELIMINA (SOFT DELETE) AL USUARIO CON ID 1
       EL USUARIO NO SE BORRA FISICAMENTE, SOLO SE MARCA COMO ELIMINADO
       ESTA SE DEBE USAR CUANDO SE DEBA DAR DE BAJA UNA CUENTA

POST http://localhost:8000/api/usuarios/1/toggle-block
     // BLOQUEA AL USUARIO SI ESTA ACTIVO, O LO DESBLOQUEA SI ESTA BLOQUEADO
       ESTA SE DEBE USAR CUANDO QUIERAS SUSPENDER O REACTIVAR UNA CUENTA SIN ELIMINARLA
